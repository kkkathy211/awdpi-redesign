{{- /* "Latest articles" on the stories page: filter buttons by category (data/story_categories.json, counts over all articles) and the newest cards. "All" shows the newest `limit` (default 12) articles; a category shows its newest `limit`. The link below goes to the list of all articles, filtered the same way. */ -}} {{- $zh := eq site.Language.Lang "zh-cn" -}} {{- $limit := .Get "limit" | default 12 | int -}} {{- $pages := partial "stories/pages.html" . -}} {{- $cats := hugo.Data.story_categories.categories -}} {{- $latest := first $limit $pages -}} {{- $top := dict -}} {{- $render := $latest -}} {{- range $cats }} {{- $these := where $pages "Params.category" .id }} {{- $top = merge $top (dict .id (first $limit $these)) }} {{- $render = $render | append (first $limit $these) }} {{- end -}} {{- $render = sort (uniq $render) "Date" "desc" -}} {{- $all := "stories/all/" | relLangURL -}}
{{- range $cats }} {{- $n := len (where $pages "Params.category" .id) }} {{- if $n }} {{- end }} {{- end }}

{{- range $render }} {{- $p := . }} {{- $tags := slice }} {{- if in $latest $p }}{{ $tags = $tags | append "all" }}{{ end }} {{- with $p.Params.category }}{{ with index $top . }}{{ if in . $p }}{{ $tags = $tags | append $p.Params.category }}{{ end }}{{ end }}{{ end }} {{ partial "stories/card.html" (dict "page" $p "tags" (delimit $tags " ") "hidden" (not (in $tags "all"))) }} {{- end }}
